cmake_minimum_required(VERSION 3.16)
project(cortex_fpga_driver LANGUAGES CXX)
add_library(cortex_fpga_driver SHARED ../src/fpga_driver.cc)
target_compile_features(cortex_fpga_driver PRIVATE cxx_std_17)
target_include_directories(cortex_fpga_driver PRIVATE ../../include)
target_compile_options(cortex_fpga_driver PRIVATE -Wall -Wextra -Werror)
target_link_libraries(cortex_fpga_driver PRIVATE pthread)
install(TARGETS cortex_fpga_driver LIBRARY DESTINATION lib)
set(CORTEX_CEPH_SOURCE_ROOT "" CACHE PATH "Optional read-only ABI2 Ceph source for adapter compile check")
if(CORTEX_CEPH_SOURCE_ROOT)
  configure_file(../../include/cortex_fpga_driver_api.h cortex/cortex_fpga_driver_api.h COPYONLY)
  configure_file(../../include/cortex_fpga_protocol.h cortex/cortex_fpga_protocol.h COPYONLY)
  configure_file(../../include/cortex_driver_api.h cortex/cortex_driver_api.h COPYONLY)
  add_library(cortex_fpga_ceph_compile OBJECT ../test/fpga_ceph_compile.cc)
  target_compile_features(cortex_fpga_ceph_compile PRIVATE cxx_std_17)
  target_compile_options(cortex_fpga_ceph_compile PRIVATE -Wall -Wextra -Werror)
  target_include_directories(cortex_fpga_ceph_compile PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR} ${CORTEX_CEPH_SOURCE_ROOT}/src
    ${CORTEX_CEPH_SOURCE_ROOT}/src/cortex/osd)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME fpga_socket_fixture COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/../test/test_fpga_socket.py $<TARGET_FILE:cortex_fpga_driver>)
endif()
